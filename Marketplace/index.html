Automate price capture directly from your Discord server:

Run this simple Python bot with a free bot token. Whenever a price broadcast containing Changes in prices for buyers is posted in your channel, it relays the raw text to a free JSONbin or your server.

# Python Discord Relay Bot
import discord, requests

intents = discord.Intents.default()
intents.message_content = True
client = discord.Client(intents=intents)

FEED_BIN_URL = "https://api.jsonbin.io/v3/b/YOUR_BIN_ID"
HEADERS = {"X-Master-Key": "YOUR_API_KEY", "Content-Type": "application/json"}

@client.event
async def on_message(message):
    if "Changes in prices for buyers" in message.content:
        payload = {"raw_markdown": message.content}
        requests.put(FEED_BIN_URL, json=payload, headers=HEADERS)
        print("Updated marketplace prices to remote feed!")

client.run("YOUR_DISCORD_BOT_TOKEN")
